################ LAYOUT GENERATION ########################

# Per-ratio-range directional expand: [minRatio, maxRatio, expandTop, expandRight, expandBottom, expandLeft]
# Ratios are width/height (>1 = landscape, <1 = portrait). First matching range wins.

EXPAND_RANGE = [0.89, 1.10, 0.05, 0.05, 0.05, 0.05]   # square-ish (1.000)
EXPAND_RANGE = [1.10, 1.60, 0.05, 0.1, 0.05, 0.1]   # landscape (1.169, 1.253, 1.337, 1.504)

# Used for any ratio not covered by an EXPAND_RANGE above: [top, right, bottom, left]
EXPAND_FALLBACK = [0.05, 0.05, 0.05, 0.05]
